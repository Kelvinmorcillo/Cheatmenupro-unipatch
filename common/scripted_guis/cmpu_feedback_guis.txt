# ============================================================
# MARKED-STATE NAME LIST REFRESH (the ONLY scripted GUI here
# that writes anything). Triggered exclusively by the Refresh
# button in gui/sakuya_main.gui. It rebuilds a UI-only
# variable list (cmpu_fb_marked_states_list) on the player's
# own country containing the currently marked states, purely
# so the status strip can display their names. It never adds,
# removes or changes states, decrees, marks or buildings.
# Own marked states are added first, then foreign-marked
# states, so the strip shows them in that order.
# ============================================================

cmpu_fb_refresh_marks = {
	scope = country
	is_shown = { is_player = yes }
	is_valid = { always = yes }
	ai_is_valid = { always = no }
	effect = {
		debug_log = "CMPU_REFRESH_EXECUTED"
		if = { limit = { variable = { name = sakuya_b6_build_sel_food_service value = 1 } } debug_log = "CMPU_STALE_FOOD_SERVICE_PRESENT" }
		if = { limit = { variable = { name = sakuya_b6_build_sel_pharmaceutical_services value = 1 } } debug_log = "CMPU_STALE_PHARMA_PRESENT" }
		if = { limit = { variable = { name = sakuya_b6_build_sel_coal_mine value = 1 } } debug_log = "CMPU_SEL_COAL_MINE" }
		if = { limit = { variable = { name = sakuya_b6_build_sel_precious_metal_mine value = 1 } } debug_log = "CMPU_SEL_PRECIOUS_METAL_MINE" }
		clear_variable_list = cmpu_fb_marked_states_list
		# Own states marked with the B6/B7 mark decree first.
		every_scope_state = {
			limit = { has_decree = decree_sakuya_mark_state }
			root = { add_to_variable_list = { name = cmpu_fb_marked_states_list target = prev } }
		}
		# Foreign states marked with the B6/B7 foreign mark variable.
		every_state = {
			limit = { has_variable = sakuya_mark_foreign_state_target }
			root = { add_to_variable_list = { name = cmpu_fb_marked_states_list target = prev } }
		}
	}
}
