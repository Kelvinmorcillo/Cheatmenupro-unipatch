# ============================================================
# CMPU UNIPOLAR PATCH - ADDITIVE FEEDBACK SCRIPT VALUES
# Read-only counters used by the feedback panel in sakuya_main.gui.
# Every script value here only reads game state (marked states,
# existing building levels) and never writes anything.
# All names are unique (cmpu_fb_ prefix).
#
# Level sums count existing levels of the given building type
# across BOTH marked target sets, mirroring the injector's two
# confirm paths:
#  - player-marked states (decree_sakuya_mark_state holders)
#  - foreign marked states (sakuya_mark_foreign_state_target)
# ============================================================

# --- Pre-initialized list variable ---
# Must exist before GetList() is called in GUI datamodel
# This ensures the list is available for reading without error
# Using {} ensures it's an empty list suitable for clear_variable_list/add_to_variable_list
cmpu_fb_marked_states_list = {}

# --- Marked target counters ---

cmpu_fb_building_count_sv = {
	value = 0
	every_scope_state = {
		limit = { has_decree = decree_sakuya_mark_state }
		every_scope_building = {
			add = 1
		}
	}
	every_state = {
		limit = { has_variable = sakuya_mark_foreign_state_target }
		every_scope_building = {
			add = 1
		}
	}
}

cmpu_fb_player_marked_state_count_sv = {
	value = 0
	every_scope_state = {
		limit = { has_decree = decree_sakuya_mark_state }
		add = 1
	}
}

cmpu_fb_foreign_marked_state_count_sv = {

	}
}
