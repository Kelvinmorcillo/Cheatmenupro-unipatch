# ============================================================
# CMPU UNIPOLAR PATCH - ADDITIVE FEEDBACK CUSTOM LOC
# Country-scope customizable localizations that pick a target
# summary key based on the current marks. Read-only: triggers
# only check state, never modify it.
# Called from localization with [GetPlayer.GetCustom('name')].
# All names are unique (cmpu_fb_ prefix).
# ============================================================

# Overall marked-target summary used by the always-visible panel.
cmpu_fb_target_summary_desc = {
	type = country

	text = {
		trigger = {
			OR = {
				any_scope_state = { has_decree = decree_sakuya_mark_state }
				any_state = { has_variable = sakuya_mark_foreign_state_target }
			}
		}
		localization_key = cmpu_fb_target_any
	}
	text = {
		fallback = yes
		localization_key = cmpu_fb_target_none
	}
}

# Population generation: which confirm path is currently possible
# and what it would honestly do (one random state vs every state).
cmpu_fb_popgen_target_desc = {
	type = country

	text = {
		trigger = {
			any_scope_state = { has_decree = decree_sakuya_mark_state }
		}
		localization_key = cmpu_fb_popgen_target_one_marked
	}
	text = {
		trigger = {
			any_scope_state = { is_homeland_of_country_cultures = root }
		}
		localization_key = cmpu_fb_popgen_target_every_homeland
	}
	text = {
		trigger = {
			any_state = {
				has_variable = sakuya_mark_foreign_state_target
				owner = {
					has_variable = sakuya_mark_country_target
				}
			}
		}
		localization_key = cmpu_fb_popgen_target_one_foreign
	}
	text = {
		fallback = yes
		localization_key = cmpu_fb_popgen_target_none
	}
}
